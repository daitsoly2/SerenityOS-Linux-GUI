# Host (Linux) build of the SerenityOS WindowServer.
#
# This reuses the unmodified compositor / window manager / IPC code and swaps
# only the pieces that talk to SerenityOS-specific kernel interfaces:
#   - main.cpp                  -> main_linux.cpp
#   - HardwareScreenBackend.cpp -> LinuxScreenBackend.cpp (SDL2)

# libdrm backs the optional KMS presenter (SERENITY_PRESENTER=drm).
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBDRM REQUIRED IMPORTED_TARGET libdrm)
# Mesa GBM + EGL back the optional GBM presenter (SERENITY_WS_BACKEND=gbm):
# GBM allocates the scanout buffers, EGL (EGL_PLATFORM_GBM_KHR) binds the DRM device.
pkg_check_modules(GBM REQUIRED IMPORTED_TARGET gbm)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
# libinput + libudev: input devices the way Mutter/KWin/wlroots do it.
pkg_check_modules(LIBINPUT REQUIRED IMPORTED_TARGET libinput)
pkg_check_modules(LIBUDEV REQUIRED IMPORTED_TARGET libudev)
# Xlib backs the optional X11 presenter (SERENITY_PRESENTER=x11): the desktop as
# one X client, so the session can run inside Xwayland under a real compositor.
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
# libdbus lets the compositor serve org.freedesktop.Notifications on the session
# bus, so Linux apps running inside the port post into Serenity notifications.
pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)

find_package(SDL2 QUIET CONFIG)
if (NOT SDL2_FOUND)
    find_package(sdl2 QUIET CONFIG)
endif()
if (SDL2_FOUND OR sdl2_FOUND)
    set(SDL2_TARGET SDL2::SDL2)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(SDL2_TARGET PkgConfig::SDL2)
endif()

set(WINDOWSERVER_DIR "${SERENITY_PROJECT_ROOT}/Userland/Services/WindowServer")

# wlroots compositor presenter (SERENITY_PRESENTER=wlroots): WindowServer embeds
# wlroots and BECOMES the compositor. The native desktop and foreign X11/Wayland
# clients are all equal scene nodes in one wlroots scene graph. Optional: hosts
# without the wlroots-0.20 dev packages keep building the unchanged KMS/SDL/X11
# paths (guarded by SERENITY_HAVE_WLROOTS_PRESENTER).
pkg_check_modules(WLROOTS QUIET IMPORTED_TARGET wlroots-0.20)
if (WLROOTS_FOUND)
    pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
    # wlroots needs xkbcommon at runtime for client keymaps; its .pc only lists it
    # under Requires.private, so link it explicitly (input needs the symbols).
    pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
    # 脏区上报用 pixman region（wlroots 的 .pc 同样只在 Requires.private 里列）
    pkg_check_modules(PIXMAN REQUIRED IMPORTED_TARGET pixman-1)
    pkg_check_modules(XCURSOR REQUIRED IMPORTED_TARGET xcursor)
    # wlroots 0.20's Requires.private headers (e.g. xcb/xcb_ewmh.h pulled in by
    # <wlr/xwayland.h>) are not propagated to the compile command for a shared
    # link, so add the prefix's general include dir explicitly.
    pkg_get_variable(WLROOTS_PREFIX wlroots-0.20 prefix)
    set(WSWLR_HOST_DIR ${CMAKE_CURRENT_SOURCE_DIR}/WSWlrBridge/host)
    list(APPEND SOURCES ${WSWLR_HOST_DIR}/wlroots_backend.c)
    # wlroots_backend.c deliberately omits unused parameter names in wl_listener
    # callbacks. Newer clang treats that as a C23 extension and -Werror turns it
    # into a hard error, so relax just that warning for this one C file.
    set_source_files_properties(${WSWLR_HOST_DIR}/wlroots_backend.c
        PROPERTIES COMPILE_DEFINITIONS WLR_USE_UNSTABLE
        COMPILE_OPTIONS "-Wno-cast-qual;-Wno-c23-extensions")
    set(WLROOTS_PRESENTER_AVAILABLE TRUE)
endif()

add_executable(serenity-window-server ${SOURCES} ${GENERATED_SOURCES})

# Make every object wait for the generated endpoint headers. Without this a
# clean parallel build can compile WindowServer sources before the IPC
# compiler has produced them.
set_source_files_properties(${SOURCES} PROPERTIES OBJECT_DEPENDS "${GENERATED_SOURCES}")

add_dependencies(serenity-window-server
    generate_WindowServerEndpoint.h
    generate_WindowClientEndpoint.h
    generate_WindowManagerServerEndpoint.h
    generate_WindowManagerClientEndpoint.h
)

target_include_directories(serenity-window-server PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}
    # TaskbarWindow.h (pulled in through WindowServer/Window.h) includes the
    # taskbar IME indicator header.
    ${CMAKE_CURRENT_SOURCE_DIR}/IMEFrontend
)

target_link_libraries(serenity-window-server PRIVATE
    LibCore
    LibFileSystem
    LibKeyboard
    LibGfx
    LibThreading
    LibIPC
    LibMain
    ${SDL2_TARGET}
    PkgConfig::LIBDRM
    PkgConfig::LIBINPUT
    PkgConfig::LIBUDEV
    PkgConfig::X11
    PkgConfig::GBM
    PkgConfig::EGL
    PkgConfig::DBUS
)

# Marks the host build so shared sources (WindowManager) can add port-only
# behaviour such as the screenshot/recording global hotkeys.
target_compile_definitions(serenity-window-server PRIVATE SERENITY_LINUX_GUI_PORT=1)

if (WLROOTS_PRESENTER_AVAILABLE)
    target_include_directories(serenity-window-server PRIVATE
        ${WSWLR_HOST_DIR} ${WLROOTS_PREFIX}/include)
    target_link_libraries(serenity-window-server PRIVATE
        PkgConfig::WLROOTS PkgConfig::WAYLAND_SERVER
        PkgConfig::XKBCOMMON PkgConfig::XCURSOR PkgConfig::PIXMAN)
    target_compile_definitions(serenity-window-server PRIVATE SERENITY_HAVE_WLROOTS_PRESENTER=1)
endif()

set_target_properties(serenity-window-server PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
